### 02.cpp 逐行注释与语法说明

```cpp
#include <iostream>
#include <cstdio>
#include <cstring>
#include <climits>

#define MAX_WORKER 1000   // 工人数组上限
#define MAX_LIST 1000     // 每个工人完成列表上限
#define MAX_ORDER 100000  // 订单数组上限

int num;      // 工人数
int limit;    // pending 队列容量上限
int N;        // 指令步数

// 工人结构体，静态全局数组存储
static struct Worker {
    int uid;                 // 工人 uid
    int type;                // 工人类型
    
    int order_id;            // 当前订单 oid（-1 表示无）
    int order_index;         // 当前订单在 orders[] 的下标（-1 表示无）
    int avai_time;           // 可用时间（-1 表示空闲；=t 表示 t 时刻完成刚好空闲）
    
    int finish_order_list[MAX_LIST]; // 已完成订单 oid 列表
    int finish_order_list_index;     // 已完成数量
} workers[MAX_WORKER];

// 订单状态枚举
enum { PENDING = 0, DONE, DOING, DISCARDED };

// 订单结构体
static struct Order {
    int oid;        // 订单编号
    int priority;   // 优先级（大者优先）
    int order_time; // 需要时长
    int type;       // 类型
    int state;      // 状态：PENDING/DONE/DOING/DISCARDED
} orders[MAX_ORDER];

static char cmd[128];                  // 命令缓存（C 风格字符串）
int oid, priority, order_time, type, uid; // 临时读入变量

static int curr_order_index;   // 当前已使用的订单数量（也是下一个可写入下标）
static int pending_orders_count;// 当前 pending 数量

// 新增订单逻辑：若 pending 满，则丢弃 pending 中最低优先级；然后插入新订单
void add_order(int oid, int priority, int order_time, int type) {
    if (pending_orders_count == limit) {
        int target_priority = INT_MAX; // 寻找最小优先级
        int target_index = -1;
        for (int i = 0 ; i < curr_order_index; ++i) {
            if (orders[i].priority < target_priority && orders[i].state == PENDING) {
                target_priority = orders[i].priority;
                target_index = i;
            }
        }
        
        orders[target_index].state = DISCARDED; // 标记被丢弃
        
        // 在末尾插入新订单
        orders[curr_order_index].oid = oid;
        orders[curr_order_index].priority = priority;
        orders[curr_order_index].order_time = order_time;
        orders[curr_order_index].type = type;
        orders[curr_order_index].state = PENDING;
        
        curr_order_index++;
    } else {
        // 直接插入新订单
        orders[curr_order_index].oid = oid;
        orders[curr_order_index].priority = priority;
        orders[curr_order_index].order_time = order_time;
        orders[curr_order_index].type = type;
        orders[curr_order_index].state = PENDING;
        
        pending_orders_count++;  
        curr_order_index++;
    }

}

// 查询工人状态：根据 avai_time 是否为 -1 判断是否休息
void queryUser(int uid) {
    for (int i = 0 ; i < num; ++i) {
        if (workers[i].uid == uid) {
            if (workers[i].avai_time == -1) {
                printf("worker %d resting\n", uid);
            } else {
                printf("worker %d doing order %d\n", uid, workers[i].order_id);
            }
        }
    }
}

// 查询订单状态
void queryOrder(int oid) {
    for (int i = 0 ; i < curr_order_index; ++i) {
        if (orders[i].oid == oid) {
            switch (orders[i].state) {
                case PENDING:
                    printf("order %d pending\n", oid);
                    break;
                case DONE:
                    printf("order %d done\n", oid);
                    break;
                case DOING:
                    printf("order %d doing\n", oid);
                    break;
                case DISCARDED:
                    printf("order %d discarded\n", oid);
                    break;
            }
        }
    }
}

// 查询某工人历史完成订单列表
void queryOrders(int uid) {
    for (int i = 0 ; i < num; ++i) {
        if (workers[i].uid == uid) {
            for (int j = 0 ; j < workers[i].finish_order_list_index; ++j) {
                printf("%d ", workers[i].finish_order_list[j]);
            }
        }
    }
    printf("\n");
}

// 选择某类型优先级最高的 pending 订单，返回其下标；无则返回 -1
int select_order(int type) {
    int target_priority = INT_MIN;
    int target_index = -1;
    
    for (int i = 0 ; i < curr_order_index; ++i) {
        Order target_order = orders[i];
        if (target_order.state == PENDING && target_order.type == type &&
            target_order.priority > target_priority) {
            target_priority = target_order.priority;
            target_index = i;
        }
    }
    
    return target_index;
}

// 每轮调度逻辑：
// - 若工人刚好在此刻完成任务，则记录完成并置为空闲；
// - 空闲工人尝试领取匹配类型、最高优先级的 pending 订单；
// - 领取后设置完成时间（当前时间 + order_time）。
void logic(int curr_time) {
    if (curr_time == 0) {
        return; // 第 0 轮不做结算（与实现约定保持）
    }
    
    for (int i = 0; i < num; ++i) {
        if (workers[i].avai_time == curr_time || workers[i].avai_time == -1) {
            if (workers[i].avai_time == curr_time) {
                workers[i].finish_order_list[workers[i].finish_order_list_index] = workers[i].order_id; // 记录完成 oid
                workers[i].finish_order_list_index++;
                
                orders[workers[i].order_index].state = DONE; // 标记订单完成
            }
            
            int order_index = select_order(workers[i].type);
            if (order_index != -1) {
                workers[i].avai_time = curr_time + orders[order_index].order_time; // 设置下一次空闲时刻
                workers[i].order_id = orders[order_index].oid;
                workers[i].order_index = order_index;
                
                orders[order_index].state = DOING;
                pending_orders_count--;
            } else {
                workers[i].avai_time = -1; // 置为空闲
                workers[i].order_id = -1;
                workers[i].order_index = -1;
            }
        }
    }
}

int main() {
    scanf("%d %d", &num, &limit); // 读入工人数与 pending 上限
    for (int i = 0; i < num; ++i) {
        scanf("%d %d", &workers[i].uid, &workers[i].type);
        workers[i].order_id = -1;      // 无当前订单
        workers[i].avai_time = -1;     // 空闲
        workers[i].order_index = -1;   // 无订单下标
    }
    scanf("%d", &N); // 读入总操作数
    for (int i = 0; i < N; ++i) {
        logic(i); // 每轮先调度
        
        scanf("%s", cmd); // 读命令
        if (!strcmp(cmd, "add")) {
            scanf("%d %d %d %d", &oid, &priority, &order_time, &type);
            add_order(oid, priority, order_time, type);
        } else if (!strcmp(cmd, "queryUser")) {
            scanf("%d", &uid);
            queryUser(uid);
        } else if (!strcmp(cmd, "queryOrder")) {
            scanf("%d", &oid);
            queryOrder(oid);
        } else if (!strcmp(cmd, "queryOrders")) {
            scanf("%d", &uid);
            queryOrders(uid);
        }
    }
}
```

#### 要点与语法说明
- 使用 C 风格 I/O（`scanf/printf`）与 C++ 并存，注意字符串命令使用 `char cmd[]` 并以 `strcmp` 比较。
- 状态管理：工人以 `avai_time==-1` 判空闲；订单状态枚举清晰，避免魔法数。
- 调度策略：空闲即抢占匹配类型中优先级最高的 pending 订单；完成时间为 `当前时间 + order_time`。
- 丢弃策略：pending 满时，丢弃 pending 中优先级最低的订单。
