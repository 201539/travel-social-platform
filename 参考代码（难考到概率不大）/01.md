### 01.cpp 逐行注释与语法说明

```cpp
#include <iostream> // 引入标准输入输出库，提供 cin / cout
#include <string>   // 引入字符串库，提供 std::string
using namespace std; // 使用标准命名空间，方便直接写 cin / cout / string

const int maxn=10010; // 定义最大数组长度为 10010（常量表达整数上限）

// 定义工人（铁匠）结构体
struct Worker{
    int uid;   // 工人唯一编号（用户ID）
    int type;  // 工人擅长的订单类型（用于匹配订单 type）
    int pos;   // 当前状态/关联的订单索引：0 表示休息，否则存放正在处理的 file[] 下标
    //0休息否则oid
    int size;  // 已完成的订单数量（用于记录完成顺序）
    //已完成的订单
}worker[maxn]; // 定义工人数组，容量 maxn

// 定义订单结构体
struct File{
    int oid;        // 订单编号
    int priority;   // 优先级（数值越大优先级越高）
    int time;       // 订单需要的处理时长（单位：时间步）
    int type;       // 订单类型（用于和工人 type 匹配）
    int pos;        // 订单状态编码：1已完成 2尚未处理(pending) 3正在处理 4丢弃
    //1已完成2尚未处理3正被处理4丢弃
    int fin;        // 完成的顺序（该工人第几个完成）
    int reachtime;  // 完成的时间点（第几轮 i 完成）
    int uid;        // 处理该订单的工人“数组下标”（非工人 uid）
}file[maxn]; // 订单数组，容量 maxn

int num,limit,n,cnt; // num=工人数，limit=缓冲上限，n=时间步数，cnt=当前已加入的订单数量
string str;          // 每步读入的命令字符串

int main(){
    cin>>num>>limit;               // 读入工人数和缓冲区上限
    for(int i=1;i<=num;i++){       // 工人下标 1..num（注意 1-based 用法）
        cin>>worker[i].uid>>worker[i].type; // 读入每个工人的 uid 与 type
        worker[i].pos=0;           // 初始均休息（pos=0）
    }
    cin>>n;                        // 读入总轮数（指令步数）
    for(int i=0;i<=n-1;i++){       // 主循环：每一轮 i
        for(int j=1;j<=cnt;j++){   // 先结算到时完成的订单
            if(file[j].reachtime==i){       // 若订单 j 的完成时间等于当前轮 i
                worker[file[j].uid].size++; // 对应工人累计完成数 +1
                file[j].fin=worker[file[j].uid].size; // 记录完成序号
                file[j].pos=1;              // 置订单为已完成
                worker[file[j].uid].pos=0;  // 工人回到休息
            }
        }
        for(int j=1;j<=num;j++){            // 给空闲工人分配订单
            if(worker[j].pos==0){           // 工人 j 空闲
                int vis=-1;                 // vis=候选订单的数组下标
                for(int k=1;k<=cnt;k++){    // 遍历查找：pending 且 类型匹配 且 优先级最高
                    if(file[k].pos==2&&file[k].type==worker[j].type){
                        if(vis==-1) vis=k;  // 初始化候选
                        else{
                            if(file[k].priority>file[vis].priority) vis=k; // 选更高优先级
                        }
                    }
                }
                if(vis!=-1){                         // 找到了可做订单
                    file[vis].pos=3;                 // 标记为 doing
                    file[vis].reachtime=i+file[vis].time; // 计算完成时间点
                    worker[j].pos=vis;               // 工人绑定该订单下标
                    file[vis].uid=j;                 // 记录工人“数组下标”（非 uid）
                }
            }
        }
        cin>>str;                           // 读入本轮命令
        if(str=="add"){                    // 新增订单
            cnt++;                          // 新订单下标 = cnt
            cin>>file[cnt].oid>>file[cnt].priority>>file[cnt].time>>file[cnt].type; // 读入
            file[cnt].pos=2;                // 初始 pending
            int cur=0;                      // 统计 pending 数
            int vis=-1;                     // 记录优先级最低的 pending 的下标
            for(int j=1;j<=cnt-1;j++){      // 在已存在订单中统计/查找
                if(file[j].pos==2)
                {
                    cur++;
                    if(vis==-1) vis=j;     // 初始化
                    else{
                        if(file[j].priority<file[vis].priority) vis=j; // 更低优先级
                    }
                }
            }
            if(cur>=limit){                 // 若 pending 达上限
                file[vis].pos=4;           // 丢弃优先级最低者
            }
        }
        else if(str=="queryUser"){        // 查询工人状态
            int Uid;
            cin>>Uid;                      // 读入工人 uid（非数组下标）
            for(int j=1;j<=num;j++){
                if(worker[j].uid==Uid){
                    if(worker[j].pos==0){
                        cout<<"worker "<<Uid<<" resting"<<endl;
                    }
                    else{
                        cout<<"worker "<<Uid<<" doing order "<<file[worker[j].pos].oid<<endl;
                    }
                }
            }
        }
        else if(str=="queryOrder"){       // 查询订单状态
            int Oid;
            cin>>Oid;
            for(int j=1;j<=cnt;j++){
                if(file[j].oid==Oid){
                    if(file[j].pos==1){
                        cout<<"order "<<file[j].oid<<" done"<<endl;
                    }
                    else if(file[j].pos==2){
                        cout<<"order "<<file[j].oid<<" pending"<<endl;
                    }
                    else if(file[j].pos==3){
                        cout<<"order "<<file[j].oid<<" doing"<<endl;
                    }
                    else if(file[j].pos==4){
                        cout<<"order "<<file[j].oid<<" discarded"<<endl;
                    }
                }
            }
        }
        else if(str=="queryOrders"){      // 查询某工人完成订单清单（按完成先后）
            int Uid;
            cin>>Uid;
            for(int j=1;j<=num;j++){
                if(worker[j].uid==Uid){
                    for(int t=1;t<=worker[j].size;t++){
                        for(int k=1;k<=cnt;k++){
                            if(file[k].uid==Uid&&file[k].pos==1&&file[k].fin==t){
                                if(k==1) cout<<file[k].oid; // 注意：这里的首元素判断并不严谨，但保持原实现
                                else cout<<" "<<file[k].oid;
                                break;
                            }
                        }
                    }
                }
            }
            cout<<endl;
        }
    }
    return 0; // 程序结束
}
```

#### 要点与语法解释
- 输入输出：`cin/cout` 搭配 `<<`/`>>`，`endl` 表示换行并刷新缓冲。
- 结构体：`struct` 仅承载数据字段，无成员函数；数组定义如 `worker[maxn]`。
- 状态编码：工人 `pos==0` 休息；订单 `pos` 分别为 1 done / 2 pending / 3 doing / 4 discarded。
- 调度顺序：每轮先结算完成，再为空闲工人分配最高优先级的匹配订单，最后处理命令。
- 丢弃策略：当 pending 数量达到 `limit`，丢弃 pending 中优先级最低的一个。
- 细节注意：`file[vis].uid` 保存的是工人数组下标（非工人 uid），`queryOrders` 逻辑按原实现可能混用 uid/下标，保持不改动。
