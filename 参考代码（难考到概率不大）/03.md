### 03.cpp 逐行注释与语法说明

```cpp
#include <iostream>
using namespace std;

// 订单类：封装状态与输出
class Order{
public:
    int oid;       // 订单编号
    int priority;  // 优先级
    int needTime;  // 需要的时间
    int type;      // 类型
    int forgeTime; // 已经消耗的时间
    int state;     // 0:pending,1:doing,2:done,3:discarded
    Order(int oid,int priority,int needTime,int type):oid(oid),priority(priority),needTime(needTime),type(type){
        forgeTime=0;   // 初始未开始
        state=0;       // 初始 pending
    }
    Order (){}
    void output(){
        if(state==0){
            cout<<"order "<<oid<<" "<<"pending"<<"\n";
        }else if(state ==1){
            cout<<"order "<<oid<<" "<<"doing"<<"\n";
        }else if(state==2){
            cout<<"order "<<oid<<" "<<"done"<<"\n";
        }else if(state==3){
            cout<<"order "<<oid<<" "<<"discarded"<<"\n";
        }
    }
    void startdoing(){
        state=1; // 标记进入 doing
    }
};

int findOrder(Order order[],int oid,int numOfOrders);              // 根据 oid 找下标
int findOrderWithType(Order order[],int type,int numOfOrders);     // 找到指定类型中优先级最高的 pending 的 oid

// 工人（铁匠）
class Forger{
public:
    int finish[100]; // 已完成订单 oid 列表
    int finishNum;   // 完成数
    int uid;         // 工人 uid
    int type;        // 工人类型
    int state;       // -1:waiting; >0:doing 的 oid
    Forger(int uid,int type):uid(uid),type(type){
        finishNum=0;
        state=-1; // 初始等待
    }
    Forger(){}
    void outPut(){
        if(state==-1){
            cout<<"worker "<<uid<<" resting"<<"\n";
        }else{
            cout<<"worker "<<uid<<" doing order "<<state<<"\n";
        }
    }
    void outPutRe(){
        if(finishNum==0){
            cout<<"\n";
        }else{
            for(int i=0;i<finishNum-1;i++){
                cout<<finish[i]<<" ";
            }
            cout<<finish[finishNum-1]<<"\n";
        }
    }
    // 每步推进：若在做则累积时间，完成后切下一个；若空闲则尝试接新订单
    void whetherKeepForge(Order order[],int numOfOrders){
        if(state!=-1){
            int orderIndex = findOrder(order,state,numOfOrders);// 当前 doing 的订单下标
            order[orderIndex].forgeTime++;
            // 判断此订单是否完成
            if(order[orderIndex].forgeTime==order[orderIndex].needTime){
                order[orderIndex].state=2;          // done
                finish[finishNum]=state;            // 记录完成 oid
                finishNum++;
                // 继续寻找下一个符合 type 的订单
                int start = findOrderWithType(order,type,numOfOrders);// 返回 oid
                // 无下一个
                if(start==-1)
                    state=-1; // 休息
                else{
                    state=start; // 切换到新订单（按 oid 标识）
                    order[findOrder(order,state,numOfOrders)].startdoing();
                }
            }
        }else{
            // 空闲则寻找下一个匹配类型的 pending 订单
            int start = findOrderWithType(order,type,numOfOrders);//oid
            if(start==-1)
                state=-1;
            else{
                state=start;
                order[findOrder(order,state,numOfOrders)].startdoing();
            }
        }
    }
};

int findIndexOfForger(Forger forgers[],int uid,int numOfForger){
    for(int i=0;i<numOfForger;i++){
        if(forgers[i].uid==uid)
            return i;
    }
    return -1;
}

void dowork(Forger forgers[],int numOfForger,Order orders[],int numOfOrders){
    for(int i=0;i<numOfForger;i++){
        forgers[i].whetherKeepForge(orders,numOfOrders);
    }
}

// 返回订单数组下标
int findOrder(Order order[],int oid,int numOfOrders){
    for(int i=0;i<numOfOrders;i++){
        if(order[i].oid==oid)
            return i;
    }
    return -1;
}

// 返回某类型、pending 状态下优先级最高的订单 oid；若无返回 -1
int findOrderWithType(Order order[],int type,int numOfOrders){
    bool first=true;
    int maxPr_oid=-1;
    for(int i=0;i<numOfOrders;i++){
        if(order[i].type==type&&order[i].state==0){
            if(first){
                first=false;
                maxPr_oid=order[i].oid;
            }else{
                if(order[findOrder(order,maxPr_oid,numOfOrders)].priority<order[i].priority){
                    maxPr_oid = order[i].oid;
                }
            }
        }
    }
    return maxPr_oid;
}

int countValid(Order orders[],int numOfOrders){
    int valid=0;
    for(int i=0;i<numOfOrders;i++){
        if(orders[i].state==0){ // pending
            valid++;
        }
    }
    return valid;
}

// 丢弃一个 pending 中优先级最低的订单（置 state=3）
void discardOne(Order orders[],int numOfOrders){
    int discardedOne=0;
    bool first=true;
    for(int i=0;i<numOfOrders;i++){
        if(orders[i].state==0){
            if(first){
                discardedOne=i;
                first=false;
            }else{
                if(orders[discardedOne].priority>orders[i].priority){
                    discardedOne=i;
                }
            }
        }
    }
    orders[discardedOne].state=3;
}

int main(){
    int numOfForger=0;
    int billLimit=0; // pending 上限
    cin>>numOfForger;
    cin>>billLimit;
    Forger forgers[numOfForger];
    for(int i=0;i<numOfForger;i++){
        int uid=0;
        int type=0;
        cin>>uid;
        cin>>type;
        Forger f(uid,type);
        forgers[i]=f;
    }
    int startTimeNum=0;
    cin>>startTimeNum; // 操作步数
    Order orders[startTimeNum]; // 预分配最多这么多订单
    int numOfOrders=0;
    string cmd;
    for(int i=0;i<startTimeNum;i++){
        dowork(forgers,numOfForger,orders,numOfOrders); // 每轮先推进工作
        cin>>cmd;
        if(cmd=="queryUser"){
            int uid=0;
            cin>>uid;
            int index = findIndexOfForger(forgers,uid,numOfForger);
            forgers[index].outPut();
        }
        else if(cmd=="add"){
            int valid=countValid(orders,numOfOrders);
            if(valid==billLimit) discardOne(orders,numOfOrders); // 若满则丢弃最低优先级 pending
            int oid=0,priority=0,needTime=0,type=0;
            cin>>oid;cin>>priority;cin>>needTime;cin>>type;
            Order o(oid,priority,needTime,type);
            orders[numOfOrders]=o;
            numOfOrders++;
        }
        else if(cmd=="queryOrder"){
            int oid=0;
            cin>>oid;
            int index = findOrder(orders,oid,numOfOrders);
            orders[index].output();
        }
        else if(cmd=="queryOrders"){
            int uid=0;
            cin>>uid;
            int index = findIndexOfForger(forgers,uid,numOfForger);
            forgers[index].outPutRe();
        }
    }
}
```

#### 要点与语法说明
- 工人与订单以类实现，封装状态与方法（如 `Order::output`）。
- 调度策略：工人持续推进当前订单，完成后马上寻找同类型优先级最高的 pending 订单。
- 丢弃策略：当 pending 数达上限，丢弃 pending 中优先级最低者。
- 本实现用 `state` 区分四种状态，保持与题面一致：0/1/2/3 = pending/doing/done/discarded。
