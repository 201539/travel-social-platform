### 07.cpp 逐行注释与语法说明

```cpp
#include <iostream>
#include <cassert>
#include <limits>

using namespace std;

#define DEBUG(...)      cout << "DEBUG: " << __VA_ARGS__ << endl
#define LINK(A, B)      (B)->next = (A)->next; (A)->next = (B)
#define UNLINK(A, B)    (A)->next = (B)->next; (B)->next = nullptr

constexpr int UNSTARTED = 0; // 未开始（在 waiting 队列）
constexpr int RUNNING   = 1; // 运行中（在各 smith 的 running 列表）
constexpr int COMPLETE  = 2; // 完成（在各 smith 的 finished 列表）
constexpr int ABORTED   = 3; // 丢弃
constexpr int ORDER_MAX_N = 100; // 最大订单节点数

int max_queue_n = 0; // waiting 队列容量上限
int waiting_n = 0;   // waiting 当前数量

class Weapon {
public:
    int oid;
    int prio;
    int time;
    int type;
    
    Weapon(int _oid, int _prio, int _time, int _type):
        oid{_oid}, prio{_prio}, time{_time}, type{_type} {
    }
};

class WpNode {
public:
    Weapon *w;  // 指向订单
    int status; // 状态
    WpNode *next; // 单链表 next
    
    WpNode(Weapon *_w, int _status): w{_w}, status{_status}
    { }
};

WpNode *nodes[ORDER_MAX_N]; // 所有节点指针的登记处（便于查询）
int wpnode_n = 0;           // 已登记节点数量

WpNode *waiting_dhead;      // waiting 队列的哨兵头（dummy head）

class Smith {
public:
    int uid;
    int type;
    
    WpNode *running_dhead;  // 正在做的链表（实际只用头->next）
    WpNode *finished_dhead; // 完成列表
    
    Smith(int _uid, int _type): uid{_uid}, type{_type}
    {
        running_dhead = new WpNode(nullptr, UNSTARTED);
        finished_dhead = new WpNode(nullptr, UNSTARTED);
    }
    
    void work() {
        WpNode *nxt;
        
        if (now_working_on() != nullptr) { // 若有工作
            nxt = running_dhead->next;
            assert(nxt->w->time >= 0);
            if (nxt->w->time == 0) { // 已完成则挪入 finished
                nxt->status = COMPLETE;
                WpNode *cur = finished_dhead;
                while (cur->next != nullptr)
                    cur = cur->next;
                UNLINK(running_dhead, cur); // 注意：原实现这里的 UNLINK(cur, nxt) 疑似笔误，这里保持原宏用法解释
                LINK(cur, nxt);
            }
        }
        
        if (now_working_on() == nullptr) { // 若空闲，从 waiting 取匹配类型且优先级最高
            nxt = waiting_dhead->next;
            WpNode *prenxt = waiting_dhead;
            if (nxt == nullptr) return;
            
            // 先找到第一个 type 匹配的
            while (nxt != NULL && nxt->w->type != type) {
                prenxt = nxt;
                nxt = nxt->next;
            }
            if (nxt == NULL) return;
            
            // 在匹配类型中选出优先级最高者
            WpNode *pre = waiting_dhead, *cur = pre->next;
            for (; cur != nullptr; pre = cur, cur = cur->next) {
                if (cur->w->type == type && cur->w->prio > nxt->w->prio) {
                    prenxt = pre;
                    nxt = cur;
                }
            }
            if (nxt == NULL) return;
            
            UNLINK(prenxt, nxt);   // 从 waiting 移除
            LINK(running_dhead, nxt); // 加入 running
            nxt->status = RUNNING;
            waiting_n--;
        }
        
        Weapon &w = (*nxt->w);
        w.time--; // 进行一单位时间
    }
    
    void queryUser() {
        cout << "worker" << uid; // 注意原输出少了空格，保持原样
        Weapon *w = now_working_on();
        if (w == nullptr)
            cout << " resting";
        else
            cout << " doing order " << w->oid;
        cout << endl;
    }
    
    void add_work(Weapon *w) {
        if (waiting_n == max_queue_n) { // 满了则丢弃 waiting 中最低优先级
            if (waiting_dhead->next == nullptr) return;
            WpNode *min_node = waiting_dhead->next,
                    *cur = min_node->next;
            for (; cur != nullptr; cur = cur->next) {
                if (cur->w->prio < min_node->w->prio)
                    min_node = cur;
            }
            for (cur = waiting_dhead; cur->next != nullptr && cur->next != min_node; cur = cur->next);
            UNLINK(cur, min_node);
            min_node->status = ABORTED;
            waiting_n--;
        }
        WpNode *node = new WpNode(w, UNSTARTED);
        nodes[wpnode_n++] = node;
        assert(wpnode_n < ORDER_MAX_N);
        LINK(waiting_dhead, node); // 头插
        waiting_n++;
    }
    
    void work_finished() {
        WpNode *cur = finished_dhead->next;
        for (; cur != nullptr; cur = cur->next)
            cout << cur->w->oid << " ";
        cout << endl;
    }
private:
    Weapon* now_working_on() {
        WpNode *first = running_dhead->next;
        if (first == nullptr) return nullptr;
        return first->w;
    }
};

// 全局 smith 序列
Smith **smith_a;
int smith_n;

Smith& find_smith(int uid) {
   for (int i = 0; i < smith_n; i++) {
        Smith &smith = (*smith_a[i]);
        if (smith.uid == uid) return smith;
    }
    assert(0);
}

Smith& find_smith_by_type(int type) {
   for (int i = 0; i < smith_n; i++) {
        Smith &smith = (*smith_a[i]);
        if (smith.type == type) return smith;
    }
    assert(0);
}

void smith_work() {
    for (int i = 0; i < smith_n; i++) {
        Smith &smith = (*smith_a[i]);
        smith.work();
    }
}

// add [oid] [priority] [time] [type]
void op_add() {
    int oid, prio, time, type;
    cin >> oid >> prio >> time >> type;
    Weapon *weapon = new Weapon(oid, prio, time, type);
    Smith &smith = find_smith_by_type(type);
    smith.add_work(weapon);
}

void op_quser() {
    int uid;
    cin >> uid;
    find_smith(uid).queryUser();
}

void op_qorder() {
    int oid;
    cin >> oid;
    cout << "order " << oid;
    for (int i = 0; i < wpnode_n; i++) {
        WpNode &wn = (*nodes[i]);
        if (wn.w->oid == oid) {
            switch (wn.status) {
                case UNSTARTED: cout << " pending";   break;
                case RUNNING:   cout << " doing";     break;
                case COMPLETE:  cout << " done";      break;
                case ABORTED:   cout << " discarded"; break;
                default: assert(0);
            }
            cout << endl;
            return;
        }
    }
    assert(0);
}

void op_qorders() {
    int uid;
    cin >> uid;
    Smith &smith = find_smith(uid);
    smith.work_finished();
}

void op_handle() {
    string cmd;
    cin >> cmd;
    if (cmd == "add")
        op_add();
    else if (cmd == "queryUser")
        op_quser();
    else if (cmd == "queryOrder")
        op_qorder();
    else if (cmd == "queryOrders")
        op_qorders();
    else{
        assert(0);
    }
}

int main() {
    cin >> smith_n >> max_queue_n;
    smith_a = new Smith*[smith_n];
    waiting_dhead = new WpNode(nullptr, UNSTARTED);
    for (int i = 0; i < smith_n; i++) {
        int uid, type;
        cin >> uid >> type;
        smith_a[i] = new Smith(uid, type);
    }
    int op_n;
    cin >> op_n;
    for (int i = 0; i < op_n; i++) {
        smith_work();
        op_handle();
    }
    return 0;
}
```

#### 要点与语法说明
- 使用单链表与哨兵头管理 waiting/running/finished，宏 `LINK/UNLINK` 简化插入/移除。
- waiting 满时丢弃最低优先级，头插入新单；smith 空闲时从 waiting 取匹配类型的最高优先级。
- 查询通过全局 `nodes[]` 登记的节点状态判断 pending/doing/done/discarded。
