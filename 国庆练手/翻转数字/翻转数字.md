#include <iostream>
#include <limits>
using namespace std;

int main() {
    int N;
    cin >> N;
    
    int sign = (N < 0) ? -1 : 1;
    long long num = static_cast<long long>(N) * sign;
    
    long long reversed = 0;
    while (num > 0) {
        reversed = reversed * 10 + num % 10;
        num /= 10;
    }
    
    reversed *= sign;
    
    if (reversed < numeric_limits<int>::min() || reversed > numeric_limits<int>::max()) {
        cout << -1 << endl;
    } else {
        cout << static_cast<int>(reversed) << endl;
    }
    
    return 0;
}
