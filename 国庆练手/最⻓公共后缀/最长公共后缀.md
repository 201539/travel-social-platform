#include <iostream>
#include <vector>
#include <string>
#include <algorithm>
using namespace std;

string longestCommonSuffix(vector<string>& strs) {
    if (strs.empty()) return "";
    
    // 找到最短字符串的长度
    int minLen = strs[0].length();
    for (const string& s : strs) {
        minLen = min(minLen, (int)s.length());
    }
    
    // 从后往前比较每个字符
    string result = "";
    for (int i = 1; i <= minLen; i++) {
        char currentChar = strs[0][strs[0].length() - i];
        bool allMatch = true;
        
        // 检查所有字符串在当前位置的字符是否相同
        for (int j = 1; j < strs.size(); j++) {
            if (strs[j][strs[j].length() - i] != currentChar) {
                allMatch = false;
                break;
            }
        }
        
        if (allMatch) {
            result = currentChar + result; // 将匹配的字符加到结果前面
        } else {
            break; // 一旦不匹配就停止
        }
    }
    
    return result;
}

int main() {
    int n;
    cin >> n;
    
    vector<string> strs(n);
    for (int i = 0; i < n; i++) {
        cin >> strs[i];
    }
    
    string result = longestCommonSuffix(strs);
    cout << result << endl;
    
    return 0;
}
